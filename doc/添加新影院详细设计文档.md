# 添加新影院系列详细设计文档

> **用途**：指导 agent 完成新影院系列（如 SFC/UME/辰星/凤凰/金逸/万达）的接入开发
> **适用场景**：新增一个影院品牌/系列的 API 接入，使其能在报价规则、影院映射、购票出票等全流程中使用

---

## 1. 两层扩展架构

本系统有两类"可扩展点"，不要混淆：

```
┌─────────────────────────────────────────────────────┐
│              票务平台层（Platform）—— 拉单 + 报价      │
│  猎人 / 省APP / 芒果 / 蚂蚁 / 洋葱 / 影划算 / 商展 ... │
│  负责：从平台拉待报价订单 → 匹配报价规则 → 提交报价      │
│  代码位置：src/common/platform/queues/               │
└──────────────────────┬──────────────────────────────┘
                       │ 订单中的 app_name 决定走哪个影院系列
                       ▼
┌─────────────────────────────────────────────────────┐
│              影院系列层（Cinema）—— 报价 + 出票        │
│  SFC / UME / LMA / 辰星 / 凤凰 / 金逸 / 万达 ...       │
│  负责：登录影院系统 → 计算报价 → 选座锁座 → 获取取票码   │
│  代码位置：src/common/autoTicket/buyTicket/           │
└─────────────────────────────────────────────────────┘
```

> **本文档聚焦影院系列层**。票务平台层的添加参考 `src/common/platform/` 下现有实现。

### 步骤 2：创建请求实例

**文件**：`src/utils/http/newcinema-request.js`

有两种典型模式：

#### 模式 A：直连后端（SFC/UME 等）

```javascript
import axios from "axios";
import { ElMessage } from "element-plus";

const createAxios = ({ app_name, timeout = 25 }) => {
  const instance = axios.create({
    baseURL: "",
    timeout: timeout * 1000
  });

  // 请求拦截器：注入 token
  instance.interceptors.request.use(config => {
    // 从 config.data 或 config.params 中提取 token，注入到 headers
    if (config.data?.session_id) {
      const sid = config.data.session_id;
      delete config.data.session_id;
      config.headers["session-id"] = sid;
    }
    return config;
  });

  // 响应拦截器：返回 response.data（统一为 .data 即为业务数据）
  instance.interceptors.response.use(
    response => response.data,
    error => {
      ElMessage.error("网络连接异常");
      return Promise.reject(error);
    }
  );

  return instance;
};

export default createAxios;
```

#### 模式 B：后端代理转发（万达等需要签名/加密的系列）

参考 `src/utils/http/wanda-film-request.js`：

```javascript
// 请求拦截器核心逻辑
instance.interceptors.request.use(config => {
  // 从请求体中提取 wanda_token，注入为请求头
  if (config.data?.wanda_token) {
    const session_id = config.data.wanda_token;
    delete config.data.wanda_token;
    config.headers["user-token"] = session_id || "";
  }
  // 同样处理 wanda_identifier → user-identifier 头
  if (config.data?.wanda_identifier) {
    config.headers["user-identifier"] = config.data.wanda_identifier;
    delete config.data.wanda_identifier;
  }
  // URL 走后端代理：/wanda-film/{alias}/path → 后端转发到万达域名
  return config;
});

// 响应拦截器：默认返回 response.data
instance.interceptors.response.use(response => response.data);
```

> **设计决策**：如果 API 需要签名/加密或后端中转，用模式 B。

#### 8.2 seatManage.js — 座位管理

```javascript
import { formatErrInfo } from "@/utils/utils";
import { APP_API_OBJ } from "@/common/index";

export default class NewcinemaSeatManage {
  constructor(order, logger, isTestOrder) {
    this.order = order;
    this.appFlag = order.app_name;
    this.logger = logger;
    this.isTestOrder = isTestOrder;
    this.appApi = APP_API_OBJ[order.app_name];
  }

  /** 获取座位图 */
  async getSeatLayout({ showTimeId, session_id }) {
    const res = await this.appApi.getRealTimeSeat(
      { dId: showTimeId },
      { data: { newcinema_token: session_id } }
    );
    return res?.data?.realtimeSeats || null;
    // 返回结构: { area: [{ areaId, areaPrice: { salesPrice }, seat: [...] }] }
  }

  /** 自动选座 */
  async autoSelectSeat(seatData, ticketNum) {
    const areas = seatData.area || [];
    for (const area of areas) {
      const seats = area.seat || [];
      const areaPrice = area.areaPrice?.salesPrice || 0;

      // ⚠️ 座位状态字段因影院而异！
      // 万达：status === 1 可售，status === 3 已售
      // SFC：座位数组中 item[4] 表示状态
      const available = seats.filter(s => s.status === 1); // ← 按实际 API 字段调整

      if (available.length >= ticketNum) {
        const selected = this.findConsecutiveSeats(available, ticketNum);
        // 注入区域价格（⚠️ 价格可能在 area.areaPrice，不在座位对象上）
        return selected.map(s => ({ ...s, salesPrice: areaPrice }));
      }
    }
    return [];
  }

  /** 查找连续座位（同排相邻优先） */
  findConsecutiveSeats(availableSeats, ticketNum) {
    const rowGroups = {};
    for (const seat of availableSeats) {
      const row = seat.row || seat.coordy;
      if (!rowGroups[row]) rowGroups[row] = [];
      rowGroups[row].push(seat);
    }
    for (const row of Object.keys(rowGroups)) {
      rowGroups[row].sort((a, b) => (a.coordx || 0) - (b.coordx || 0));
    }
    for (const row of Object.keys(rowGroups)) {
      const seats = rowGroups[row];
      for (let i = 0; i <= seats.length - ticketNum; i++) {
        const slice = seats.slice(i, i + ticketNum);
        const isConsecutive = slice.every(
          (s, j) => j === 0 || s.coordx - slice[j - 1].coordx === 1
        );
        if (isConsecutive) return slice;
      }
    }
    return availableSeats.slice(0, ticketNum); // 退化为取前N个
  }

  getSeatPrice(seatInfo) {
    return seatInfo?.salesPrice || 0;
  }
}
```

**⚠️ 两个高频坑**：

1. **座位状态字段名**：必须根据实际 API 响应确认（`status` / `seatStatus` / `seat_state`）
2. **价格来源**：可能在座位对象上，也可能在 `area.areaPrice` 上

#### 8.3 orderManage.js — 订单管理

```javascript
import { formatErrInfo } from "@/utils/utils";
import { APP_API_OBJ } from "@/common/index";

export default class NewcinemaOrderManage {
  constructor(order, logger, platManage, isTestOrder) {
    this.logger = logger;
    this.platManage = platManage; // ← 用于转单操作
    this.order = order;
    this.appFlag = order.app_name;
    this.appApi = APP_API_OBJ[order.app_name];
    this.isTestOrder = isTestOrder;

  async createOrder({ dId, retailerCode, mobile, seatId, session_id }) {
    const res = await this.appApi.createOrder(
      { dId, retailerCode, mobile, seatId, json: true },
      { data: { newcinema_token: session_id } }
    );

  /** 确认订单（绑定手机号） */

    const res = await this.appApi.confirmOrder(
      { orderId, mobilePhone, json: true },
      { data: { newcinema_token: session_id } }
    );
    return res?.data || null;
  }

  /** 查询订单状态 */
  async queryOrderStatus({ orderId, session_id }) {
    const res = await this.appApi.queryOrderStatus(
      { orderId },
      { data: { newcinema_token: session_id } }
    );
    return res?.data || null;
  }

  /** 取消订单（释放座位） */
  async cancelOrder({ orderId, session_id }) {
    const res = await this.appApi.cancelOrder(
      { orderId, json: true },
      { data: { newcinema_token: session_id } }
    );
    return res?.data || null;
  }

  /** 转单 — 通知平台转给其他供应商 */
  async transferOrder(extra = {}) {
    return await this.platManage.orderTransferByPlat(
      extra?.reason || "出票失败",
      extra
    );
  }
}
```

**关键点**：

- `createOrder` 的 `seatId` 格式：`"seatId,price,undefined,0|seatId2,price2,undefined,0"`
- `retailerCode` 通常是 `"MX"`
- ⚠️ 转单方法名是 `orderTransferByPlat(errMsg, errInfo)`，不是 `transferOrderByPlat`

#### 8.5 offerManage.js — 报价管理（★最关键）

**继承 `BaseOfferPrice`，实现报价管线。** 基类已实现 `getEndOfferPrice()` 模板方法，子类需实现：

| 方法                                     | 职责                                                                      |
| ---------------------------------------- | ------------------------------------------------------------------------- |
| `initModules(order)`                     | 初始化 logger、cardQuanManage、cinemaManage、seatManage                   |
| `getEndMatchOfferRule(order)`            | 调用 `offerRuleMatch()` 匹配规则，再调用 `getMinAmountOfferRule()` 选最优 |
| `getMinAmountOfferRule(ruleList, order)` | 从规则列表中选最低价：会员日优先 → 固定报价 → 会员价加价                  |
| `calculateFinalPrice(params)`            | **5步价格管线**：动态调价 → 利润加价 → 夜间顶价 → 超限检查 → 成本利润     |

**报价管线 5 步**（函数来自 `../common/offerHelper`）：

| 步骤 | 函数                                                                                         | 逻辑                                                       |
| :--: | -------------------------------------------------------------------------------------------- | ---------------------------------------------------------- |
|  1   | `applyDynamicPricing({ basePrice, offerList, logger })`                                      | 读 `localStorage.adjustPrice`，达到进单/出单阈值时 ±N 元   |
|  2   | `applyProfitAddition({ price, offerType, appFlag, groupList, logger })`                      | 读 `localStorage.profitAddPrice`，非固定报价时额外加价     |
|  3   | `applyNightMaxPrice({ price, supplier_max_price, logger })`                                  | 读 `localStorage.isOpenisNightMaxPrice`，凌晨1-6点顶到限价 |
|  4   | `handleOverrunCheck({ price, supplier_max_price, plat_name, logger })`                       | 超限价时：允许则压回限价，不允许返回 null                  |
|  5   | `calcOfferCostProfitParts({ adjustedPrice, cost_price, rewards, plat_name, noFeePlatList })` | 手续费(1%) - 奖励金 + 成本 = 真实成本，利润≤0不报价        |

**模板代码**：

```javascript
import { offerRuleMatch, formatErrInfo } from "@/utils/utils";
import { APP_API_OBJ } from "@/common/index.js";
import { NO_FEE_PLAT_LIST } from "@/common/constant.js";
import Logger from "@/common/logger.js";
import BaseOfferPrice from "@/common/core/BaseOfferPrice.js";
import NewcinemaCardQuanManage from "./cardQuanManage.js";
import NewcinemaCinemaManage from "./cinemaManage.js";
import NewcinemaSeatManage from "./seatManage.js";
import {
  applyDynamicPricing,
  applyProfitAddition,
  applyNightMaxPrice,
  handleOverrunCheck,
  calcOfferCostProfitParts
} from "../common/offerHelper";

class NewcinemaOfferPrice extends BaseOfferPrice {
  constructor({ appFlag, plat_name }) {
    super({ appFlag, plat_name });
    this.appApi = APP_API_OBJ[appFlag];
  }

  initModules(order) {
    this.logger = new Logger({ logType: 1 });
    this.logger.init(order);
    this.cardQuanManage = new NewcinemaCardQuanManage(order, this.logger);
    this.cinemaManage = new NewcinemaCinemaManage(order, this.logger);
    this.seatManage = new NewcinemaSeatManage(order, this.logger, false);
  }

  async getEndMatchOfferRule(order) {
    const matchRuleListRes = offerRuleMatch(order, this.logger);
    let matchRuleList = matchRuleListRes?.matchRuleList || [];
    if (!matchRuleList?.length) return null;
    return await this.getMinAmountOfferRule(matchRuleList, order);
  }

  async getMinAmountOfferRule(ruleList, order) {
    // 1. 会员日规则优先（offerType=3 + memberDay）
    const memberDayRules = ruleList.filter(
      item => item.memberDay && item.offerType === "3" && item.offerAmount
    );
    memberDayRules.sort((a, b) => a.offerAmount - b.offerAmount);
    if (memberDayRules.length) return memberDayRules[0];

    // 2. 固定报价（offerType=1）→ 按金额取最低
    const otherRules = ruleList.filter(
      item => !item.memberDay && item.offerType !== "3"
    );
    const fixedRules = otherRules.filter(
      item => item.offerType === "1" && item.offerAmount
    );
    if (fixedRules.length) {
      fixedRules.sort((a, b) => a.offerAmount - b.offerAmount);
      return fixedRules[0];
    }

    // 3. 会员价加价（offerType=2）
    const addAmountRules = otherRules.filter(
      item => item.offerType === "2" && item.addAmount
    );
    if (addAmountRules.length) return addAmountRules[0];

    return null;
  }

  async calculateFinalPrice(params) {
    const {
      cost_price,
      supplier_max_price,
      price,
      rewards,
      offerType,
      offerList,
      offerRule
    } = params;
    try {
      // 1. 动态调价
      let adjusted = applyDynamicPricing({
        basePrice: price,
        offerList,
        logger: this.logger
      });
      // 2. 利润加价
      adjusted = applyProfitAddition({
        price: adjusted,
        offerType,
        appFlag: this.appFlag,
        groupList: [],
        logger: this.logger
      });
      // 3. 夜间顶价
      adjusted = applyNightMaxPrice({
        price: adjusted,
        supplier_max_price,
        logger: this.logger
      });
      // 4. 超限检查
      adjusted = await handleOverrunCheck({
        price: adjusted,
        supplier_max_price,
        plat_name: this.plat_name,
        logger: this.logger
      });
      if (!adjusted) return null;
      // 5. 成本利润核算
      const parts = calcOfferCostProfitParts({
        adjustedPrice: adjusted,
        cost_price,
        rewards,
        plat_name: this.plat_name,
        noFeePlatList: NO_FEE_PLAT_LIST
      });
      if (Number(parts.expectProfit) <= 0) {
        this.logger.errorSave("预期利润不足", parts);
        return null;
      }
      return Number(adjusted);
    } catch (error) {
      this.logger.errorSave("计算报价异常", { error: formatErrInfo(error) });
      return null;
    }
  }
}

export default NewcinemaOfferPrice;
```

#### 8.6 buyTicket.js — 出票主流程（★最关键）

**继承 `BaseBuyTicket`。** 基类 `singleTicket()` 已实现完整模板，子类只需实现：

| 方法                      | 职责                                                                     |
| ------------------------- | ------------------------------------------------------------------------ |
| `initModules()`           | 实例化 platManage、seatManage、orderManage、cinemaManage、cardQuanManage |
| `getCinemaLoginInfo()`    | 从 localStorage 筛选该影线的可用登录信息                                 |
| `oneClickBuyTicket(item)` | **核心：9步出票流程**                                                    |

**出票流程 9 步**：

```
① 验证影院存在     → cinemaManage.getCinemaDetail()
② 匹配影片         → cinemaManage.getMovieInfo()
③ 获取排期         → cinemaManage.getShowTimeList()
④ 获取座位图+选座  → seatManage.getSeatLayout() + autoSelectSeat()
⑤ 组装 seatId 字符串 → "seatId,price,undefined,0|..."
⑥ 创建订单（锁座） → orderManage.createOrder()
⑦ 确认订单         → orderManage.confirmOrder()
⑧ 轮询获取取票码   → orderManage.queryOrderStatus()
⑨ 上传取票码       → platManage.submitTicketCode()
```

**失败处理**：步骤 ①②③④⑥⑧ 失败时 → 调用 `orderManage.transferOrder()` 转单。

**模板代码**：

```javascript
import { mockDelay } from "@/utils/utils";
import BaseBuyTicket from "@/common/core/BaseBuyTicket.js";
import NewcinemaSeatManage from "./seatManage.js";
import NewcinemaOrderManage from "./orderManage.js";
import NewcinemaCinemaManage from "./cinemaManage.js";
import NewcinemaCardQuanManage from "./cardQuanManage.js";
import PlatManage from "../platManage.js";

class NewcinemaBuyTicket extends BaseBuyTicket {
  initModules() {
    this.platManage = new PlatManage(this.order, this.logger);
    this.seatManage = new NewcinemaSeatManage(
      this.order,
      this.logger,
      this.isTestOrder
    );
    this.orderManage = new NewcinemaOrderManage(
      this.order,
      this.logger,
      this.platManage,
      this.isTestOrder
    );
    this.cinemaManage = new NewcinemaCinemaManage(this.order, this.logger);
    this.cardQuanManage = new NewcinemaCardQuanManage(this.order, this.logger);
  }

  async getCinemaLoginInfo() {
    this.currentParamsList = this.getLoginInfoList().filter(
      item => item.app_name === this.appFlag && item.mobile && item.session_id
    );
    this.currentSessionId = this.currentParamsList[0]?.session_id || "";
    this.currentPhone = this.currentParamsList[0]?.mobile || "";
  }

  async oneClickBuyTicket(item) {
    const { order_number, film_name, show_time, ticket_num, otherParams } =
      item;
    let { offerRule, city_id, cinema_id } = otherParams || {};

    // ① 验证影院存在
    const cinemaInfo = await this.cinemaManage.getCinemaDetail({
      storeId: cinema_id,
      session_id: this.currentSessionId
    });
    if (!cinemaInfo)
      return this.orderManage.transferOrder({ reason: "获取影院信息失败" });

    // ② 匹配影片
    const movieInfo = await this.cinemaManage.getMovieInfo(this.order);
    if (!movieInfo)
      return this.orderManage.transferOrder({ reason: "未匹配到电影" });

    // ③ 获取排期
    const showTimeList = await this.cinemaManage.getShowTimeList({
      cinemaId: cinema_id,
      filmId: movieInfo.film_id,
      date: show_time?.split(" ")[0],
      session_id: this.currentSessionId
    });
    if (!showTimeList)
      return this.orderManage.transferOrder({ reason: "获取排期失败" });

    // ④ 获取座位图 + 自动选座
    const showTimeId = showTimeList.showtimeList?.[0]?.id;
    const seatLayout = await this.seatManage.getSeatLayout({
      showTimeId,
      session_id: this.currentSessionId
    });
    const targetSeats = await this.seatManage.autoSelectSeat(
      seatLayout,
      ticket_num || 1
    );
    if (!targetSeats.length)
      return this.orderManage.transferOrder({ reason: "无可用座位" });

    // ⑤ 组装 seatId 字符串
    const seatIdStr = targetSeats
      .map(s => `${s.seatId},${s.salesPrice},undefined,0`)
      .join("|");

    // ⑥ 创建订单（锁座）
    const orderRes = await this.orderManage.createOrder({
      dId: showTimeId,
      retailerCode: "MX",
      mobile: this.currentPhone,
      seatId: seatIdStr,
      session_id: this.currentSessionId
    });
    if (!orderRes)
      return this.orderManage.transferOrder({ reason: "锁座失败" });

    // ⑦ 确认订单
    await this.orderManage.confirmOrder({
      orderId: orderRes.orderId,
      mobilePhone: this.currentPhone,
      session_id: this.currentSessionId
    });

    // ⑧ 轮询取票码（最多10次，每次间隔2秒）
    for (let i = 0; i < 10; i++) {
      await mockDelay(2);
      const status = await this.orderManage.queryOrderStatus({
        orderId: orderRes.orderId,
        session_id: this.currentSessionId
      });
      if (status?.orderStatus === "SUCCESS" || status?.ticketCode) {
        // ⑨ 上传取票码到平台
        const submitRes = await this.platManage.submitTicketCode(
          order_number,
          status.ticketCode
        );
        return {
          profit: 0,
          submitRes,
          qrcode: status.ticketCode,
          offerRule
        };
      }
    }
    return this.orderManage.transferOrder({ reason: "获取取票码超时" });
  }
}

export default NewcinemaBuyTicket;
```

## 5. 报价规则类型说明

| offerType | 含义       | 成本价来源                        | 适用场景             |
| :-------: | ---------- | --------------------------------- | -------------------- |
|   `"1"`   | 固定报价   | `offerRule.offerAmount`（DB配置） | 券/会员卡固定价格    |
|   `"2"`   | 会员价加价 | 影院会员价 + `addAmount`          | 会员卡余额支付赚差价 |
|   `"3"`   | 会员日报价 | `offerRule.offerAmount`           | 会员日特价           |

**规则匹配优先级**：会员日(3) > 固定报价(1) > 会员价加价(2)
