# 批量同步新平台报价规则 实施计划

> **Goal:** 新增平台时，以已有平台报价为参考，批量将所有匹配规则中的报价原样复制到新平台，无需逐条手动编辑。

**Architecture:** 后端新增单条 CASE WHEN 批量 SQL（每次最多 300 条），前端在报价规则页面增加"批量同步新平台"按钮和参数选择对话框。

**Tech Stack:** Node.js + MySQL（raw SQL）、Vue 3 + Element Plus + Pinia

```js
    // 批量追加平台报价（用于新平台上线时批量同步报价规则）
    batchAddPlatOffer: async ({
        sourcePlatName,    // 参考平台 "lieren"
        targetPlatName,    // 目标平台 "piaosheng"
        appType,           // 可选过滤：影线类型
        appName,           // 可选过滤：影线
        statusList         // 可选过滤：规则状态 ["1","3"]
    }) => {
        try {
            // 1. 查询匹配规则
            let selectSql = `SELECT id, platOfferList, orderForm FROM rule_record WHERE NOT status = '4'`;
            const selectParams = [];

            if (appType) {
                selectSql += ` AND app_type = ?`;
                selectParams.push(appType);
            }
            if (appName) {
                selectSql += ` AND shadowLineName = ?`;
                selectParams.push(appName);
            }
            if (statusList?.length) {
                selectSql += ` AND status IN (${statusList.map(() => '?').join(',')})`;
                selectParams.push(...statusList);
            }

            const rules = await baseService.query(selectSql, selectParams);
            console.log(`batchAddPlatOffer 查询到 ${rules.length} 条规则`);

            // 2. 内存计算
            const toUpdate = [];
            let skippedCount = 0;
            let noSourceCount = 0;

            for (const rule of rules) {
                let platList;
                try {
                    platList = JSON.parse(rule.platOfferList || "[]");
                } catch {
                    platList = [];
                }

                // 已存在目标平台 → 跳过
                if (platList.some(p => p.platName === targetPlatName)) {
                    skippedCount++;
                    continue;
                }

                // 找到参考平台报价
                const sourceOffer = platList.find(p => p.platName === sourcePlatName);
                if (!sourceOffer) {
                    noSourceCount++;
                    continue;
                }

                // 追加新平台（原样复制价格）
                platList.push({
                    platName: targetPlatName,
                    value: sourceOffer.value,
                    isSyncPlat: "2"
                });

                const newOrderForm = platList.map(p => p.platName).join(",");
                toUpdate.push({
                    id: rule.id,
                    platOfferList: JSON.stringify(platList),
                    orderForm: newOrderForm
                });
            }

            console.log(`batchAddPlatOffer 待更新${toUpdate.length}条, 已有跳过${skippedCount}条, 无参考平台跳过${noSourceCount}条`);

            if (!toUpdate.length) {
                return successRes({
                    updatedCount: 0,
                    skippedCount,
                    noSourceCount,
                    totalChecked: rules.length
                }, "无规则需要更新");
            }

            // 3. 分批执行批量 CASE WHEN UPDATE（每批300条）
            const BATCH_SIZE = 300;
            let updatedCount = 0;

            for (let i = 0; i < toUpdate.length; i += BATCH_SIZE) {
                const chunk = toUpdate.slice(i, i + BATCH_SIZE);
                const ids = [];
                const platCases = [];
                const orderCases = [];
                const updateParams = [];

                for (const item of chunk) {
                    ids.push(item.id);
                    platCases.push(`WHEN ? THEN ?`);
                    updateParams.push(item.id, item.platOfferList);
                    orderCases.push(`WHEN ? THEN ?`);
                    updateParams.push(item.id, item.orderForm);
                }

                const batchSql = `
                    UPDATE rule_record
                    SET
                        platOfferList = CASE id ${platCases.join(" ")} END,
                        orderForm = CASE id ${orderCases.join(" ")} END,
                        update_time = NOW()
                    WHERE id IN (${ids.map(() => '?').join(",")})
                `;

                const res = await baseService.query(batchSql, [...updateParams, ...ids]);
                updatedCount += chunk.length;
                console.log(`batchAddPlatOffer 批次${Math.floor(i / BATCH_SIZE) + 1} 完成, 更新${chunk.length}条`);
            }

            return successRes({
                updatedCount,
                skippedCount,
                noSourceCount,
                totalChecked: rules.length
            }, `批量同步完成：更新${updatedCount}条，已有跳过${skippedCount}条，无参考平台跳过${noSourceCount}条`);

        } catch (error) {
            console.error("批量追加平台报价异常", error);
            return failRes(null, error?.sqlMessage);
        }
    },
```

```js
// 批量同步新平台报价
const handleBatchSyncNewPlat = () => {
  ElMessageBox.confirm(
    `<div>
      <p style="margin-bottom:12px">请选择参考平台和目标平台：</p>
      <div style="margin-bottom:8px">
        <span>参考平台：</span>
        <select id="batchSourcePlat" style="padding:4px 8px;width:150px">
          ${Object.entries(orderFormObj.value)
            .map(([k, v]) => `<option value="${k}">${v}</option>`)
            .join("")}
        </select>
      </div>
      <div style="margin-bottom:8px">
        <span>目标平台：</span>
        <select id="batchTargetPlat" style="padding:4px 8px;width:150px">
          ${Object.entries(orderFormObj.value)
            .map(([k, v]) => `<option value="${k}">${v}</option>`)
            .join("")}
        </select>
      </div>
      <p style="color:#e6a23c;font-size:12px;margin-top:8px">将参考平台的报价<b>原样复制</b>到目标平台，跳过已有目标平台的规则</p>
    </div>`,
    "批量同步新平台报价",
    {
      dangerouslyUseHTMLString: true,
      confirmButtonText: "开始同步",
      cancelButtonText: "取消",
      type: "warning",
      beforeClose: async (action, instance, done) => {
        if (action !== "confirm") {
          done();
          return;
        }
        const sourcePlat = document.getElementById("batchSourcePlat")?.value;
        const targetPlat = document.getElementById("batchTargetPlat")?.value;
        if (!sourcePlat || !targetPlat) {
          ElMessage.warning("请选择参考平台和目标平台");
          done();
          return;
        }
        if (sourcePlat === targetPlat) {
          ElMessage.warning("参考平台和目标平台不能相同");
          done();
          return;
        }
        batchSyncing.value = true;
        try {
          const res = await svApi.batchAddPlatOffer({
            sourcePlatName: sourcePlat,
            targetPlatName: targetPlat,
            appType: formData.app_type || undefined,
            appName: formData.shadowLineName || undefined,
            statusList: ["1", "3"]
          });
          if (res.code === 200) {
            ElMessage.success(
              res.msg || `更新${res.data?.updatedCount || 0}条规则`
            );
            searchData();
          } else {
            ElMessage.error(res.msg || "同步失败");
          }
        } catch (err) {
          ElMessage.error("同步异常");
        } finally {
          batchSyncing.value = false;
          done();
        }
      }
    }
  );
};
```
