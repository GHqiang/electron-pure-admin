> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增「单量分析」页，按平台 / 系列 / 影院三维展示报价量、报价成功量、出票量、出票成功量，一眼看出哪里的单子多。

**Architecture:** 后端在 `api/statisticalAnalysis.js` 新增 `aggregateCross`，按 `(app_name, plat_name)` 分组返回原始计数（历史读 `stat_daily_record`、今天读实时表、空表回退全实时）；前端一次请求拿全量行，在 `utils/aggregate.js` 纯函数里聚合成三个视图，系列映射复用登录后已有的影院列表（`app_type_code`）。不改表结构、不改现有页面。

**Tech Stack:** Koa2 + MySQL（后端 `auto-ticket-service`）、Vue 3 + Element Plus + Pinia（前端 `electron-pure-admin-fix`）、Jest 双端单测。

**设计文档：** `doc/plans/2026-10-21-平台系列单量分析设计.md`

- [ ] **Step 3: 实现 `aggregateCross`**

在 `auto-ticket-service/api/statisticalAnalysis.js` 中，`aggregatePlat` 的结束 `},` 之后、`// 按日预聚合` 注释之前，插入（4 空格缩进）：

```js
    // 平台×影院交叉聚合:按 (app_name, plat_name) 分组(历史读预聚合表 + 今天实时)
    // 供「单量分析」页使用:一次返回影院粒度原始计数,系列/平台维度由前端聚合
    aggregateCross: async (obj) => {
        try {
            const { start_time, end_time } = obj;
            const FIELDS = ["offer_success_num", "offer_fail_num", "ticket_total_num", "ticket_success_num"];
            // 复合键合并:mergeAgg 只保留分组键,这里需要同时保留 app_name / plat_name
            const SEP = "\u0001";
            const map = new Map();
            const addRows = (rows) => {
                (rows || []).forEach(row => {
                    const key = `${row.app_name}${SEP}${row.plat_name}`;
                    let item = map.get(key);
                    if (!item) {
                        item = { app_name: row.app_name, plat_name: row.plat_name };
                        FIELDS.forEach(f => { item[f] = 0; });
                        map.set(key, item);
                    }
                    FIELDS.forEach(f => { item[f] += Number(row[f] || 0); });
                });
            };
            // 实时表三查并行(失败报价排除入队前丢弃 err_type>=16)
            const addRealtimeRows = async (whereObj) => {
                const { whereSql, params } = buildWhere(whereObj);
                const [o1, o2, t1] = await Promise.all([
                    baseService.query(
                        `SELECT app_name, plat_name, COUNT(*) AS offer_success_num
                         FROM offer_record${whereSql}
                         GROUP BY app_name, plat_name`, params),
                    baseService.query(
                        `SELECT app_name, plat_name, COUNT(*) AS offer_fail_num
                         FROM offer_record_fail${withRealFailFilter(whereSql)}
                         GROUP BY app_name, plat_name`, params),
                    baseService.query(
                        `SELECT app_name, plat_name,
                                COUNT(*) AS ticket_total_num,
                                SUM(CASE WHEN order_status = '1' THEN 1 ELSE 0 END) AS ticket_success_num
                         FROM ticket_record${whereSql}
                         GROUP BY app_name, plat_name`, params),
                ]);
                addRows(o1);
                addRows(o2);
                addRows(t1);
            };

            const daily = buildDailyWhere(obj);
            let histRows = [];
            if (daily.whereSql) {
                histRows = await baseService.query(
                    `SELECT app_name, plat_name,
                            SUM(offer_success_num) AS offer_success_num,
                            SUM(offer_fail_num) AS offer_fail_num,
                            SUM(ticket_total_num) AS ticket_total_num,
                            SUM(ticket_success_num) AS ticket_success_num
                     FROM stat_daily_record${daily.whereSql}
                     GROUP BY app_name, plat_name`, daily.params);
            }
            // 空表回退:历史部分应有数据但预聚合表为空 → 全范围实时(与 query/plat 行为一致)
            if (histRows.length === 0 && daily.whereSql && (await dailyRangeEmpty(daily))) {
                await addRealtimeRows(obj);
                return [...map.values()];
            }
            addRows(histRows);
            const range = realtimeRange(start_time, end_time);
            if (range) {
                await addRealtimeRows({ ...obj, start_time: range.start, end_time: range.end });
            }
            return [...map.values()];
        } catch (error) {
            handleError("查询平台×影院交叉聚合异常", error);
        }
    },
```

```js
/**
 * 单量分析聚合工具(纯函数,无 Vue 依赖,便于单测)
 * 入参 rows 为后端 /statisticalAnalysis/cross 的原始行:
 *   { app_name, plat_name, offer_success_num, offer_fail_num, ticket_total_num, ticket_success_num }
 */

export const UNKNOWN_SERIES_CODE = "unknown";
export const UNKNOWN_SERIES_NAME = "未分类";

const METRIC_FIELDS = [
  "offerTotalNum",
  "offerSuccessNum",
  "ticketTotalNum",
  "ticketSuccessNum"
];

const toNum = value => {
  const n = Number(value);
  return Number.isFinite(n) ? n : 0;
};

// 成功率:分母 <= 0 返回 null(页面显示 "--"),避免"没单"与"全失败"混淆
export const rateOf = (numerator, denominator) =>
  denominator > 0 ? numerator / denominator : null;

// 系列索引:appTypeList = [{ app_type_code, app_type_name, app_name_list: [] }]
export const buildSeriesIndex = (appTypeList = []) => {
  const map = new Map();
  (appTypeList || []).forEach(item => {
    ((item && item.app_name_list) || []).forEach(appName => {
      map.set(appName, {
        code: item.app_type_code,
        name: item.app_type_name || item.app_type_code
      });
    });
  });
  return map;
};

// 取系列信息,未知影院归「未分类」
export const seriesOf = (seriesIndex, appName) =>
  (seriesIndex && seriesIndex.get(appName)) || {
    code: UNKNOWN_SERIES_CODE,
    name: UNKNOWN_SERIES_NAME
  };

// 单行 -> 四指标(缺字段按 0)
export const metricsOf = row => {
  const offerSuccessNum = toNum(row && row.offer_success_num);
  const offerFailNum = toNum(row && row.offer_fail_num);
  return {
    offerTotalNum: offerSuccessNum + offerFailNum,
    offerSuccessNum,
    ticketTotalNum: toNum(row && row.ticket_total_num),
    ticketSuccessNum: toNum(row && row.ticket_success_num)
  };
};

const emptyMetrics = () =>
  METRIC_FIELDS.reduce((acc, key) => {
    acc[key] = 0;
    return acc;
  }, {});

const addMetrics = (target, source) => {
  METRIC_FIELDS.forEach(key => {
    target[key] += toNum(source && source[key]);
  });
  return target;
};

// 平台列顺序:已知平台按 platOrder 排在前,未知平台按出现顺序追加
const orderPlats = (plats, platOrder = []) => {
  const known = platOrder.filter(plat => plats.has(plat));
  const unknown = [...plats].filter(plat => !platOrder.includes(plat));
  return [...known, ...unknown];
};

// 平台×系列交叉:行=系列,列=平台,单元格=当前指标值
export const buildCrossView = (
  rows = [],
  { seriesIndex, platOrder = [], metric = "offerTotalNum" } = {}
) => {
  const seriesMap = new Map();
  const plats = new Set();
  const unmatchedCinemas = new Set();

  (rows || []).forEach(row => {
    if (!row || !row.app_name) return;
    const series = seriesOf(seriesIndex, row.app_name);
    if (series.code === UNKNOWN_SERIES_CODE) unmatchedCinemas.add(row.app_name);
    const plat = row.plat_name || "";
    plats.add(plat);
    let seriesItem = seriesMap.get(series.code);
    if (!seriesItem) {
      seriesItem = {
        code: series.code,
        name: series.name,
        cells: {},
        metrics: emptyMetrics()
      };
      seriesMap.set(series.code, seriesItem);
    }
    const metrics = metricsOf(row);
    addMetrics(seriesItem.metrics, metrics);
    seriesItem.cells[plat] = (seriesItem.cells[plat] || 0) + metrics[metric];
  });

  const platList = orderPlats(plats, platOrder);
  const seriesRows = [...seriesMap.values()]
    .map(item => ({ ...item, total: item.metrics[metric] || 0 }))
    .sort((a, b) => b.total - a.total);
  const colTotals = platList.reduce((acc, plat) => {
    acc[plat] = seriesRows.reduce((sum, item) => sum + (item.cells[plat] || 0), 0);
    return acc;
  }, {});
  // 每列最大值:交叉表热力底色按列归一化
  const maxCell = platList.reduce((acc, plat) => {
    acc[plat] = seriesRows.reduce((max, item) => Math.max(max, item.cells[plat] || 0), 0);
    return acc;
  }, {});
  const grandTotal = seriesRows.reduce((sum, item) => sum + item.total, 0);

  return {
    seriesRows,
    plats: platList,
    colTotals,
    grandTotal,
    maxCell,
    unmatchedCinemas: unmatchedCinemas.size
  };
};

// 系列排行:按当前指标降序,附双成功率与占总量比
export const buildSeriesRank = (
  rows = [],
  { seriesIndex, metric = "offerTotalNum" } = {}
) => {
  const map = new Map();
  (rows || []).forEach(row => {
    if (!row || !row.app_name) return;
    const series = seriesOf(seriesIndex, row.app_name);
    let item = map.get(series.code);
    if (!item) {
      item = { code: series.code, name: series.name, metrics: emptyMetrics() };
      map.set(series.code, item);
    }
    addMetrics(item.metrics, metricsOf(row));
  });
  const list = [...map.values()];
  const total = list.reduce((sum, item) => sum + item.metrics[metric], 0);
  return list
    .map(item => ({
      code: item.code,
      name: item.name,
      ...item.metrics,
      offerRate: rateOf(item.metrics.offerSuccessNum, item.metrics.offerTotalNum),
      ticketRate: rateOf(item.metrics.ticketSuccessNum, item.metrics.ticketTotalNum),
      share: total > 0 ? item.metrics[metric] / total : 0
    }))
    .sort((a, b) => b[metric] - a[metric]);
};

// 影院排行:影院级聚合,附系列名与名称回退
export const buildCinemaRank = (
  rows = [],
  { seriesIndex, appLabels = {}, metric = "offerTotalNum", limit = null } = {}
) => {
  const map = new Map();
  (rows || []).forEach(row => {
    if (!row || !row.app_name) return;
    let item = map.get(row.app_name);
    if (!item) {
      const series = seriesOf(seriesIndex, row.app_name);
      item = {
        appName: row.app_name,
        name: appLabels[row.app_name] || `${row.app_name}-已调整`,
        seriesCode: series.code,
        seriesName: series.name,
        metrics: emptyMetrics()
      };
      map.set(row.app_name, item);
    }
    addMetrics(item.metrics, metricsOf(row));
  });
  const list = [...map.values()]
    .map(item => ({
      ...item,
      ...item.metrics,
      offerRate: rateOf(item.metrics.offerSuccessNum, item.metrics.offerTotalNum),
      ticketRate: rateOf(item.metrics.ticketSuccessNum, item.metrics.ticketTotalNum)
    }))
    .sort((a, b) => b[metric] - a[metric]);
  return limit && limit > 0 ? list.slice(0, limit) : list;
};

// 周期小结:四指标合计 + 双成功率
export const buildSummary = (rows = []) => {
  const metrics = emptyMetrics();
  (rows || []).forEach(row => addMetrics(metrics, metricsOf(row)));
  return {
    ...metrics,
    offerRate: rateOf(metrics.offerSuccessNum, metrics.offerTotalNum),
    ticketRate: rateOf(metrics.ticketSuccessNum, metrics.ticketTotalNum)
  };
};

// 千分位数字,空值按 0
export const formatNum = value => {
  const n = Math.round(toNum(value));
  return n.toLocaleString("zh-CN");
};

// 成功率百分比,空值(--)
export const formatRate = value =>
  value === null || value === undefined ? "--" : `${(value * 100).toFixed(1)}%`;
```

```vue
const seriesOptions = computed(() =>
  GET_ALL_APP_TYPE_LIST().filter(item => (item.app_name_list || []).length > 0)
);

const userList = ref([]);
const rawRows = ref([]);
const loadError = ref("");
const activeTab = ref("cross");
const metric = ref("offerTotalNum");
const quickActive = ref(7);
// 平台/系列多选在前端过滤:后端只做等值过滤,多选循环请求代价高
const platFilter = ref([]);
const seriesFilter = ref([]);

const metricLabel = computed(() => {
  const target = METRIC_OPTIONS.find(item => item.value === metric.value);
  return target ? target.label : "单量";
});

// 表单只放需要下发给后端的参数(平台/系列多选由前端过滤)
const formData = reactive({
  user_id: "",
  start_time: "",
  end_time: ""
});
const pad2 = n => String(n).padStart(2, "0");
const getDayStart = offset => {
  const now = new Date();
  const day = new Date(now.getFullYear(), now.getMonth(), now.getDate() + offset);
  return `${day.getFullYear()}-${pad2(day.getMonth() + 1)}-${pad2(day.getDate())} 00:00:00`;
};
formData.start_time = getDayStart(-6);
formData.end_time = getDayStart(1);

// 系列索引:app_name -> { code, name }
const seriesIndex = computed(() => buildSeriesIndex(seriesOptions.value));

// 影院过滤集合:选系列时收敛到该系列下的影院
const seriesAppSet = computed(() => {
  if (!seriesFilter.value.length) return null;
  const set = new Set();
  seriesOptions.value.forEach(item => {
    if (!seriesFilter.value.includes(item.app_type_code)) return;
    (item.app_name_list || []).forEach(appName => set.add(appName));
  });
  return set;
});

const filteredRows = computed(() =>
  rawRows.value.filter(row => {
    if (platFilter.value.length && !platFilter.value.includes(row.plat_name)) return false;
    if (seriesAppSet.value && !seriesAppSet.value.has(row.app_name)) return false;
    return true;
  })
);

const summary = computed(() => buildSummary(filteredRows.value));
const crossView = computed(() =>
  buildCrossView(filteredRows.value, {
    seriesIndex: seriesIndex.value,
    platOrder: orderFormKeys,
    metric: metric.value
  })
);
const seriesRank = computed(() =>
  buildSeriesRank(filteredRows.value, {
    seriesIndex: seriesIndex.value,
    metric: metric.value
  })
);
const cinemaRank = computed(() =>
  buildCinemaRank(filteredRows.value, {
    seriesIndex: seriesIndex.value,
    appLabels: APP_LIST.value,
    metric: metric.value
  })
);

// 时间范围上限:实时表保留 30 天,超出部分由预聚合表(stat_daily_record)兜底
const MAX_QUERY_DAYS = 31;
const confirmQueryRange = async () => {
  if (!formData.start_time || !formData.end_time) return true;
  const rangeDays = (new Date(formData.end_time) - new Date(formData.start_time)) / 86400000;
  if (rangeDays <= MAX_QUERY_DAYS) return true;
  ElMessageBox.alert(
    `查询时间范围不能超过 ${MAX_QUERY_DAYS} 天(报价记录仅保留 ${MAX_QUERY_DAYS} 天,超出部分无数据)`,
    "提示",
    { type: "warning", confirmButtonText: "知道了" }
  );
  return false;
};

const quickRange = days => {
  quickActive.value = days;
  formData.start_time = days === 1 ? getDayStart(0) : getDayStart(-(days - 1));
  formData.end_time = getDayStart(1);
  loadData();
};
const onSearch = () => {
  quickActive.value = null;
  loadData();
};
const resetForm = () => {
  platFilter.value = [];
  seriesFilter.value = [];
  formData.user_id = "";
  formData.start_time = getDayStart(-6);
  formData.end_time = getDayStart(1);
  quickActive.value = 7;
  loadData();
};

// 请求竞态:序号守卫 + AbortController(与统计分析页一致)
let abortController = null;
let reqSeq = 0;
const loadData = async () => {
  if (!(await confirmQueryRange())) return;
  const seq = ++reqSeq;
  abortController?.abort();
  abortController = new AbortController();
  const signal = abortController.signal;
  const loading = ElLoading.service({
    lock: true,
    text: "Loading",
    background: "rgba(0, 0, 0, 0.7)"
  });
  try {
    // 与现有统计页一致:空值参数不下发(rule 可能为空)
    const queryParams = Object.fromEntries(
      Object.entries({
        start_time: formData.start_time,
        end_time: formData.end_time,
        user_id: formData.user_id,
        rule
      }).filter(([, value]) => value !== null && value !== undefined && value !== "")
    );
    const res = await svApi.queryAnalysisCross(queryParams, { signal });
    if (seq !== reqSeq) return;
    rawRows.value = res.data?.list || [];
    loadError.value = "";
  } catch (error) {
    if (error?.name === "CanceledError" || error?.code === "ERR_CANCELED") return;
    if (seq !== reqSeq) return;
    rawRows.value = [];
    loadError.value = error?.message || "单量数据加载失败,请重试";
  } finally {
    loading.close();
  }
};

// 系列排行点行:切到影院榜并带上该系列筛选
const onSeriesSelect = code => {
  seriesFilter.value = [code];
  activeTab.value = "cinema";
};
// 影院点行:复用现有影线明细页
const onCinemaDrill = appName => {
  router.push({ path: "/set/todayStatistics/detail", query: { app_name: appName } });
};

loadData();
onBeforeMount(async () => {
  const res = await svApi.getUserList();
  userList.value = res.data?.userList || [];
});
</script>
```

```markdown
## 核心变更说明

1. 后端新增「影院×平台」交叉聚合接口：历史读 `stat_daily_record`、今天读三张实时表、预聚合表为空时回退全实时，口径与现有 `query`/`plat` 完全一致（失败报价排除 `err_type>=16`）。
2. 前端只发一次请求，拿影院粒度原始计数后在纯函数里聚合成平台×系列、系列排行、影院排行三个视图；系列映射复用登录后已加载的影院列表（`app_type_code → APP_TYPE_OBJ`），未匹配影院归「未分类」并提示。
3. 指标口径：报价量 = 报价成功 + 报价失败；出票量 = 出票记录笔数；成功量为对应成功笔数；全部按订单笔数，不乘票数。

- 风险点：`stat_daily_record` 若未按 `stat_date` 建索引，30 天查询会变慢（需单独提 DDL，不在本次代码改动内）。
```
